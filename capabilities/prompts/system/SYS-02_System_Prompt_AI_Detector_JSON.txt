import json
import anthropic

client = anthropic.Anthropic()

def detect_ai(text):
    response = client.messages.create(
        model="claude-sonnet-4-6",
        max_tokens=1000,
        system=SYSTEM_PROMPT,  # content of this file
        messages=[{"role": "user", "content": text}]
    )
    result = json.loads(response.content[0].text)
    return result

result = detect_ai("Your text here...")
print(f"AI: {result['ai_percentage']}% — {result['recommendation']}")
print(f"Detected words: {result['signals']['ai_vocabulary_found']}")
