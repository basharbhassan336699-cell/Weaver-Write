━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
CORE TECHNICAL PRINCIPLES
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. PERPLEXITY — how predictable a text is to a language model.
   AI text tends to pick the statistically most likely next word,
   producing low perplexity. Human writing is less predictable.
   > Documented limit: OpenAI shut down its own classifier in 2023
     after it correctly identified only 26% of AI text while
     falsely flagging 9% of human text.

2. BURSTINESS — variation in sentence length across a document.
   Humans mix short and long sentences; AI output is more uniform.

3. TRAINED CLASSIFIERS — Transformer models (BERT, RoBERTa,
   DeBERTa) fine-tuned on large paired human/AI datasets.

4. CROSS-PERPLEXITY — using two LLMs at once to score
   predictability more robustly (e.g., the Binoculars method).

5. N-GRAM & VOCABULARY ANALYSIS — word/phrase frequency,
   lexical diversity, and overused AI-signature phrasing.

6. MODEL FINGERPRINTING — identifying signatures of specific
   generative models from recurring patterns.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
DETECTION LAYERS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
Character-level · Sentence-level (with highlighting) ·
Semantic coherence · Adversarial training · OCR ·
Multi-modal (image/video/code) · Similarity databases ·
User feedback loops.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
PROCESSING PIPELINE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
1. Text extraction (paste / upload / URL / OCR)
2. Feature extraction (perplexity, burstiness, vocab)
3. Initial classification (per sentence/paragraph)
4. Deeper analysis (semantic, fingerprint, adversarial)
5. Aggregation into a document-level score
6. Presentation (percentage + optional explanation)

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
ACCURACY — WHAT THE NUMBERS MEAN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
- Internally reported accuracy: 94–99.6% (optimistic, controlled data)
- Independently studied accuracy: 70–96% (varies by text/edits/language)
- False Positive Rate: <1% to >25% — the most important practical metric
- Accuracy drops sharply on paraphrased, edited, or very short texts.

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
SHARED LIMITATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
- Bias against non-native language speakers
- Unreliable on texts under ~100–200 words
- Weak on humanized / heavily edited text
- Higher false positives on specialized technical writing
- Struggles with mixed human+AI documents
- Lags behind each new generation of models

━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
THE ARMS-RACE DYNAMIC
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
As detectors improve, generation models adapt. No detector is
permanently reliable; results are an initial SIGNAL, not proof,
and must always be paired with human judgment.
